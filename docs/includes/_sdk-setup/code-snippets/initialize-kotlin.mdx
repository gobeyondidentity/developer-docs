```kotlin
// --- Initialize with required arguments
EmbeddedSdk.init(
  app = this,
  logger = { logMessage ->
    Log.d("BeyondIdentityLog", logMessage)
  }
)

// --- Initialize with required and optional arguments
EmbeddedSdk.init(
  app = this,
  logger = { logMessage ->
    Log.d("BeyondIdentityLog", logMessage)
  },
  biometricAskPrompt = "Please provide your biometric",
  allowedDomains = listOf("example.com", "another-example.com")
)

// --- Initialize from an Activity
class MainActivity : AppCompatActivity() {
  override fun onCreate(savedInstanceState: Bundle?) {
    super.onCreate(savedInstanceState)
    EmbeddedSdk.init(
      app = application,
      logger = { logMessage ->
        Log.d("BeyondIdentityLog", logMessage)
      }
    )
    // Hand the SDK the Activity that called init()
    EmbeddedSdk.setActivity(this)
  }
}
```
